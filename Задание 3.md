```
workspace {

    model {

        user = person "Пользователь"
        
        mobileApp = softwareSystem "Mobile App" "Мобильное приложение" "iOS / Android"

        backend = softwareSystem "Backend Platform" {

            apiGateway = container "API Gateway" "Точка входа" "Kong / Nginx"

            cartService = container "Cart Service" "Работа с корзиной" "Java / Kotlin"

            orderService = container "Order Service" "Работа с заказами" "Java / Kotlin"

            marketingService = container "Marketing Service" "Маркетинговые кампании" "Java / Kotlin"

            notificationService = container "Notification Service" "Отправка PUSH уведомлений" "Java / Kotlin"

            messageBroker = container "Message Broker" "Асинхронный обмен событиями" "Kafka / RabbitMQ"

            notificationDb = container "Notification DB" "Хранение токенов и шаблонов" "PostgreSQL"
            
            notificationDeamon = container "Notification Daemon" "Выполняет задачи на отправку уведомлений"
            
            fcmAdapter = container "Firebase Cloud Messaging Adapter" "Формирует сообщения для FCM"
            
            apnsAdapter = container "Apple APNs Adapter" "Формирует сообщения для APNs"
        }

        firebase = softwareSystem "Firebase Cloud Messaging"
        apns = softwareSystem "Apple APNs"

        mobileApp -> apiGateway "REST API" "Отправляет device token"
        
        user -> mobileApp "Использует"

        apiGateway -> cartService "Запросы корзины"
        apiGateway -> orderService "Запросы заказов"
        apiGateway -> messageBroker "Передаёт данные о Польователе"

        cartService -> messageBroker "Публикует cart-events"
        orderService -> messageBroker "Публикует order-events"
        marketingService -> messageBroker "Публикует marketing-events"

        notificationService -> messageBroker "Подписывается на события"
        notificationService -> messageBroker "Публикует tasks"
        notificationService -> notificationDb "Чтение/запись"
        
        notificationDeamon -> messageBroker "Подписывается на tasks"
        notificationDeamon -> fcmAdapter "Отправка push"
        notificationDeamon -> apnsAdapter "Отправка push"

        firebase -> mobileApp "Push delivery"
        apns -> mobileApp "Push delivery"
    }
}
```
